local function triggerAction()
    canBoost = false
    local att = Instance.new("Attachment", root)
    local vel = Instance.new("LinearVelocity", att)
    vel.Attachment0 = att
    vel.MaxForce = 35000
    vel.VectorVelocity = (root.CFrame.LookVector * speed) + Vector3.new(0, 12, 0)
    task.wait(delayTime)
    vel:Destroy()
    att:Destroy()
end

hum.AnimationPlayed:Connect(function(track)
    local animId = track.Animation.AnimationId:match("%d+")
    
    if animId and EmoteStuff[animId] then
        canBoost = true
        lastTriggerTime = os.clock()
    end
    
    if track.Animation.AnimationId == AverageJump then
        if canBoost and (os.clock() - lastTriggerTime) <= windowTime then
            triggerAction()
        else
            canBoost = false
        end
    end
end)
